class RoutedBankConnection(http.client.HTTPSConnection):
    def __init__(self, requested_host, **kwargs):
        super().__init__(requested_host, **kwargs)
        if self.host != host or self.port != 443:
            raise ValueError('Only the documented bank host and TLS port may use this route')
        def connect(address, timeout=socket._GLOBAL_DEFAULT_TIMEOUT, source_address=None):
            if address != (host, 443):
                raise ValueError('Proxy/tunnel host differs from the documented bank origin')
            return socket.create_connection((bank_ip,443), timeout, source_address)
        self._create_connection = connect

class BankHandler(urllib.request.HTTPSHandler):
    def https_open(self, req):
        parsed = urllib.parse.urlsplit(req.full_url)
        if parsed.scheme != 'https' or parsed.hostname != host or parsed.port not in (None,443):
            raise ValueError('Bank-only transport refuses a different origin')
        return self.do_open(RoutedBankConnection,req,context=context)

bank_open = urllib.request.build_opener(NoRedirect(),BankHandler()).open

class TimedClient(FreyaClient):
    def request(self, method, path, body=None, params=None):
        start = time.monotonic()
        result = super().request(method,path,body,params)
        item = {'method':method,'route_kind':'platform','elapsed_seconds':round(time.monotonic()-start,3)}
        if '/iterations/' in path and isinstance(result,dict) and isinstance(result.get('runs'),list):
            item.update(route_kind='iteration_page',simulation_id=path.split('/')[4],iteration_id=path.split('/')[6],
                        offset=(params or {}).get('offset'),runs=len(result['runs']),total=result.get('runsTotal'))
        timings.append(item)
        return result
    def run_simulation(self, agent_id,name,variables_by_persona,max_seconds=90,concurrency=2):
        event('simulation_about_to_start', agent_id=agent_id, canonical_lease=str(self.lease_path), personas=len(variables_by_persona))
        return super().run_simulation(agent_id,name,variables_by_persona,90,concurrency)

class WitnessBackend(HostedBackend):
    def __init__(self):
        super().__init__(client,deadline,opener=self.open_bank,reserve=300)
        self.freshness=[]
        self.pending=[]
        self.forwards=[]
    def open_bank(self,req,timeout):
        start=time.monotonic()
        try:
            response=bank_open(req,timeout=timeout)
        except BaseException as exc:
            timings.append({'route_kind':'bank','method':req.get_method(),'elapsed_seconds':round(time.monotonic()-start,3),'failure_type':type(exc).__name__})
            raise
        timings.append({'route_kind':'bank','method':req.get_method(),'http_status':response.status,'elapsed_seconds':round(time.monotonic()-start,3)})
        return response
    def practice_session(self,seed):
        session=super().practice_session(seed)
        witness={'bank_case_id':seed,'variables_sha256':digest(session),
                 'customer_fingerprint':hashlib.sha256(session['customer_id'].encode()).hexdigest(),
                 'session_field_present':'session_id' in session}
        if 'session_id' in session:
            witness['session_fingerprint']=hashlib.sha256(session['session_id'].encode()).hexdigest()
        self.freshness.append(witness)
        self.pending.append(digest(session))
        return session
    def simulate(self,agent_id,name,variables,concurrency=2):
        actual=sorted(digest(v) for v in variables.values())
        match=actual==sorted(self.pending)
        self.forwards.append({'repetition':len(self.forwards)+1,'server_variables_forwarded_unchanged':match,'personas':len(variables)})
        self.pending=[]
        if not match:
            raise ValueError('Practice server variables changed before simulation')
        return super().simulate(agent_id,name,variables,concurrency)
